CREATE DATABASE canal_anonimo;

CREATE TABLE critica (
    id INT AUTO_INCREMENT PRIMARY KEY,
    setor VARCHAR(20),
    data DATETIME DEFAULT CURRENT_TIMESTAMP,
    assunto VARCHAR(100),
    descricao VARCHAR(700),
    sugestao VARCHAR(400),
    nome VARCHAR(50)
);

CREATE TABLE denuncia (
	id INT AUTO_INCREMENT PRIMARY KEY,
	tipo_denuncia VARCHAR(30),
    setor VARCHAR(20),	
	data DATETIME DEFAULT CURRENT_TIMESTAMP,
	local VARCHAR(30),
	pessoa_envolvidas VARCHAR(200),
	descricao VARCHAR(800),
    testemunhas VARCHAR(400),
	nome VARCHAR(50),
	email VARCHAR(50)
);

CREATE TABLE reclamacao (
	id INT AUTO_INCREMENT PRIMARY KEY,
	categoria VARCHAR(25),
	setor VARCHAR(20),
	data DATETIME DEFAULT CURRENT_TIMESTAMP,
	nivel_urgencia VARCHAR(5),	
	resumo_reclamacao VARCHAR(100),
	descricao VARCHAR(800),
	tentou_resolver VARCHAR(3)    	
	sugestao VARCHAR(400),	
	nome VARCHAR(50)
);

CREATE TABLE administrador (
    id INT AUTO_INCREMENT PRIMARY KEY,
    email VARCHAR(100) NOT NULL UNIQUE,
    senha_hash VARCHAR(100) NOT NULL
)
